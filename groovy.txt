pipeline {
    agent any

    stages {

        stage('Clone Repo') {
            steps {
                git branch: 'main',
                url: 'https://github.com/donskytech/sample-static-node-express-web-application.git'
            }
        }

        stage('Check Files') {
            steps {
                sh 'ls -la'
            }
        }

        stage('Create Dockerfile') {
            steps {
                sh '''
cat > Dockerfile <<EOF
FROM node:18

WORKDIR /app

COPY . .

RUN npm install

EXPOSE 3000

CMD ["node", "index.js"]
EOF
'''
            }
        }

        stage('Build Docker Image') {
            steps {
                sh 'docker build -t ramkaje341/nodeapp3:latest .'
            }
        }

        stage('Push Docker Image') {
            steps {
                sh 'docker push ramkaje341/nodeapp3:latest'
            }
        }

        stage('Create Kubernetes Manifest') {
            steps {
                sh '''
cat > deployment.yaml <<EOF
apiVersion: apps/v1
kind: Deployment

metadata:
  name: nodeapp

spec:
  replicas: 1

  selector:
    matchLabels:
      app: nodeapp

  template:
    metadata:
      labels:
        app: nodeapp

    spec:
      containers:
      - name: nodeapp
        image: ramkaje341/nodeapp3:latest

        ports:
        - containerPort: 3000

---
apiVersion: v1
kind: Service

metadata:
  name: nodeapp-service

spec:
  type: NodePort

  selector:
    app: nodeapp

  ports:
  - port: 3000
    targetPort: 3000
    nodePort: 30007
EOF
'''
            }
        }

        stage('Deploy to Kubernetes') {
            steps {
                withKubeConfig(credentialsId: 'kube_config1') {
                    sh 'kubectl apply -f deployment.yaml'
                }
            }
        }

        stage('Verify Deployment') {
            steps {
                withKubeConfig(credentialsId: 'kube_config1') {
                    sh 'kubectl get pods'
                    sh 'kubectl get svc'
                }
            }
        }
    }
}
